O programa mantém, junto de cada ponto, os expoentes de \(g\) e \(h\) que o produziram.

Dica 1 — Estado mínimo
Armazene \((x,\alpha,\beta)\) com \(x=g^\alpha h^\beta\).
Dica 2 — Resolva a congruência por completo
Quando o coeficiente não for invertível, teste todas as classes produzidas pelo mdc.
Dica 3 — Teste cada classe candidata
Uma colisão resolve uma congruência, não necessariamente uma classe única; verifique \(g^t=h\) para eliminar as demais.
Ver solução completa

Uma implementação direta usa a função por partes (5.41), avança a tartaruga uma vez e a lebre duas, e na colisão resolve \((\delta-\beta)t\equiv\alpha-\gamma\pmod{p-1}\). Este programa é executável com node rho-dlp.mjs; usa BigInt, resolve congruências mesmo quando o coeficiente não é invertível e verifica todas as classes candidatas.

const mod = (a, n) => ((a % n) + n) % n;

function powMod(a, e, n) {
  let r = 1n;
  a = mod(a, n);
  while (e > 0n) {
    if (e % 2n === 1n) r = r * a % n;
    a = a * a % n;
    e /= 2n;
  }
  return r;
}

function egcd(a, b) {
  if (b === 0n) return [a, 1n, 0n];
  const [d, x, y] = egcd(b, a % b);
  return [d, y, x - (a / b) * y];
}

function solveLinear(a, b, n) {
  a = mod(a, n); b = mod(b, n);
  const [d, inverseFactor] = egcd(a, n);
  if (b % d !== 0n) return [];
  const reducedModulus = n / d;
  const first = mod(inverseFactor * (b / d), reducedModulus);
  const answers = [];
  for (let j = 0n; j < d; j++) answers.push(first + j * reducedModulus);
  return answers;
}

function step(state, p, g, h) {
  const q = p - 1n;
  if (3n * state.x < p) {
    return { x: state.x * g % p, a: (state.a + 1n) % q, b: state.b };
  }
  if (3n * state.x < 2n * p) {
    return { x: state.x * state.x % p, a: 2n * state.a % q,
      b: 2n * state.b % q };
  }
  return { x: state.x * h % p, a: state.a, b: (state.b + 1n) % q };
}

function rhoDlp(p, g, h) {
  const q = p - 1n;
  const start = { x: 1n, a: 0n, b: 0n };
  let slow = step(start, p, g, h);
  let fast = step(step(start, p, g, h), p, g, h);
  let iteration = 1;
  while (slow.x !== fast.x && iteration <= 2 * Number(q)) {
    slow = step(slow, p, g, h);
    fast = step(step(fast, p, g, h), p, g, h);
    iteration++;
  }
  if (slow.x !== fast.x) throw new Error("colisão não encontrada");
  const coefficient = mod(fast.b - slow.b, q);
  const rhs = mod(slow.a - fast.a, q);
  const divisor = egcd(coefficient, q)[0];
  const candidates = solveLinear(coefficient, rhs, q)
    .filter((t) => powMod(g, t, p) === h);
  return { iteration, divisor, candidates };
}

for (const input of [
